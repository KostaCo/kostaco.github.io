\section{Prebrojavanja. Osnovni principi prebrojavanja. (Uopšteni) Dirihleov princip.}
Prebrojati skup znači odrediti njegovu kardinalnost.\\

\noindent Osnovni principi prebrojavanja:
\begin{enumerate}
    \item \textbf{Princip jednakosti}: Ukoliko za skupove $A$ i $B$ postoji bijekcija $f: A \to B$, tada važi \[|A| = |B|\]
    \item \textbf{Princip zbira}: Ako su konačni skupovi $A_1, A_2, \ldots, A_n$ disjunktni, tada važi \[\bigg|\bigcup_{i = 0}^n A_i \bigg| = \sum_{i = 0}^{n} | A_i |\]
    \item \textbf{Princip proizvoda}: Ako su $A$ i $B$ proizvoljni konačni skupovi, tada važi
    \[ |A \times B| = |A| \cdot |B| \]
    Važi i opštije tvrđenje. Neka je \[S \subseteq A \times B,\;\; v_a = \{(a, b) \in S \;|\; b \in B\}\; \text{i} \; k_a = \{(a, b) \in S \;|\; a \in A\}.\]
    Tada važi 
    \[ |S| = \sum_{a \in A} |v_a| = \sum_{b \in B} |k_b|\]
\end{enumerate}

\begin{itemize}
    \item \textbf{Dirihleov princip}: Ako je $k$ objekata raspoređeno u $n < k$ kutija, tada u bar jednoj kutiji ima bar 2 objekta.
    \item \textbf{Uopšteni Dirihleov princip}: Ako je $k$ objekata raspoređeno u $n$ kutija i $k > cn$, tada u bar jednoj kutiji ima bar $c + 1$ objekat.
\end{itemize}
\subsection{Ramzijevi brojevi}
\begin{itemize}
    \item \textbf{Ramzijev broj} $R(n, k)$ označava najmanji skup osoba takav da u njemu postoji ili $k$ osoba koje se poznaju ili $n$ osoba koje se ne poznaju. Zapravo, $R(n, k)$ predstavlja najmanji broj $m$ td. svako bojenje grafa $K_m$ sa dve boje sadrži potpun monohromatski podgraf sa $n$ ili $k$ čvorova. Samo nekoliko vrednosti je poznato, jer je algoritam eksponencijalne složenosti. Npr, $R(3, 3) = 6$.
\end{itemize}

\section{Binomni koeficijenti. Osnovna svojstva. Binomna i polinomna formula. }

\[
    {{n}\choose k} = \frac{n(n-1)(n-2)\dots(n-k+1)}{k!}, \;\; k \in \mathbb{N}, \; n \in \mathbb{R}, \;\; {{n}\choose 0} = 1
\]\\

Za $k \leq n$ važi
\[
    {{n}\choose k} = \frac{n!}{(n-k)!k!}
\]

\begin{theorem} \textbf{(Uslov simetričnost):}
    Za $k, n \in \mathbb{N}, k \leq n$ važi
    \[
        {{n} \choose k} = {{n} \choose {n-k}}
    \]
\end{theorem}
\begin{proof}
    \[
        {{n} \choose k} = \frac{n!}{k!(n - k)!} = \frac{n!}{(n-k)! (n - (n-k))!} = {{n} \choose {n-k}}
    \]
\end{proof}

\begin{theorem}
    \textbf{(Negacija gornjeg indeksa:)}
     Za $k \in \mathbb{N}, n \in \mathbb{R}$ važi
     \[ {{-n} \choose k} = (-1)^k {{n + k - 1} \choose k} \]
\end{theorem}
\begin{proof}
    \[
        {{-n} \choose k} = \frac{(-n)(-n - 1)\ldots (-n - k + 1)}{k!} = 
    \]
    \[
        =(-1)^k \frac{n(n-1)\ldots(n + k - 1)}{k!} = (-1)^k {{n + k - 1} \choose k}
    \]
\end{proof}

\begin{theorem}
    \textbf{(Adiciona formula):} Za $k \in \mathbb{N}$ i $n \in \mathbb{R}$ važi
    \[ {{n} \choose k} = {{n - 1} \choose k} + {{n - 1} \choose {k - 1}} \]
\end{theorem}
\begin{proof}
    \[
        {{n - 1} \choose k} + {{n - 1} \choose {k - 1}} = \frac{(n - 1)\ldots (n - k)}{k!} + \frac{(n - 1)\ldots (n - k + 1)}{(k - 1)!}
    \]
    \[
        = \frac{(n - 1)\ldots (n - k) + k(n - 1)\ldots (n - k + 1)}{k!} = \frac{(n - 1)\ldots(n - k + 1)(n - k + k)}{k!} = {{n} \choose k}
    \]
\end{proof}

\subsection{Binomna i polinomna formula}
\begin{theorem}
    \textbf{(Binomna formula):} Za $n \in \mathbb{N}_0,\; x,y \in \mathbb{R}$ važi
    \[
        (x + y)^n = \sum_{k = 0}^{n} {{n} \choose k}x^k y^{n - k}
    \]
\end{theorem}
\begin{proof}
    Indukcija po $n$. Za $n = 0$ je $1 = (x+y)^n = 1$. Pretpostavimo da formula važi za $n - 1$. Tada dobijamo
    \[
        \begin{aligned}
            (x + y)^n &= (x+y) \sum_{k = 0}^{n - 1}{{n - 1} \choose k}x^k y^{n - 1 - k}\\
            &= \sum_{k = 0}^{n-1}{{n - 1} \choose k}x^{k + 1}y^{n - 1 - k} + \sum_{k = 0}^{n - 1} {{n - 1} \choose k}x^k y^{n - k} \\
            &= x^n + \sum_{k = 1}^{n-1}{{n - 1} \choose {k-1}}x^{k + 1}y^{n - 1 - k} + y^n + \sum_{k = 1}^{n - 1} {{n - 1} \choose k}x^k y^{n - k} \\
            &= x^n + y^n + \sum_{k = 1}^{n - 1}\bigg({{n - 1} \choose k-1} + {{n - 1} \choose {k}} \bigg) x^k y ^{n-k} \\
            &= x^n + y^n + \sum_{k = 1}^{n - 1}{{n} \choose k}x^k y^{n - k}\\ 
            &= {{n} \choose n} x^n + {{n} \choose 0} y^n + \sum_{k = 1}^{n - 1}{{n} \choose k}x^k y^{n - k} \\
            &= \sum_{k = 0}^{n}{{n} \choose k}x^k y^{n - k}
        \end{aligned}
    \]
\end{proof}

\begin{theorem}
    \textbf{(Polinomna formula):} Za $n \in \mathbb{N}_0$ i $x_1, \ldots, x_r \in \mathbb{R}$ važi
    \[ \bigg(\sum_{k = 1}^{r}x_k\bigg)^n = \sum_{\substack{k_1 + k_2 + \dots + k_r = n \\ k_1, k_2, \dots, k_r \geq 0}} {{n}\choose k_1, k_2, \dots, k_r} x_1^{k_1} x_2^{k_2} \dots x_r^{k_r},\]
   gde je \[ {{n}\choose k_1, k_2, \dots, k_m} = \frac{n!}{k_1! k_2! \dots k_m!}. \]
\end{theorem}
\begin{proof}
    Dokaz se može izvesti indukcijom po $r$.
\end{proof}

\section{Binomni identiteti.}

\begin{theorem}
    \textbf{(Izvlačenje ispred zagrade):} Za $k, n \in \mathbb{N}$ važi:
    \[ {{n} \choose k} = \frac{n}{k} {{n - 1} \choose {k - 1}}, \; {{n} \choose {k}} = \frac{n}{n-k} {{n - 1} \choose k} \]
\end{theorem}
\begin{proof}
    Za $k > n$ oba identiteta slede direktno.
\end{proof}

\begin{theorem}
    \textbf{(Pojednostavljivanje proizvoda):} Za $k, n, r \in \mathbb{N}_0$ važi
    \[ {{n} \choose k + r}{{k + r} \choose k} = {{n} \choose k}{{n - k} \choose r} \]
\end{theorem}
\begin{proof}
    Dobija se direktno.
\end{proof}

\begin{theorem}
    \textbf{(Sumacione formule):} Za $r, n \in \mathbb{N}_0$ važi
    \[ \sum_{k= 0}^{n}{{r + k} \choose k} = {{n + r + 1}\choose n}, \;\;\;   \sum_{k= 0}^{n}{{k} \choose r} = {{n + 1}\choose r + 1}\]
\end{theorem}
\begin{proof}
    Identiteti se dokazuju indukcijom.\\
    Dokažimo prvo prvi identitet. Za $n = 0$ imamo $1 = 1$. Pretpostavimo da važi za $n - 1$.
    \[ \sum_{k = 0}^{n} {{r + k} \choose k} = \sum_{k = 0}^{n - 1}{{r + k} \choose k} + {{r + n} \choose n} = {{r + n} \choose {n - 1}} + {{r + n} \choose n} = {{n + r + 1} \choose n} \] \\

    \noindent Dokažimo sada drugi identitet. Za $n = 0$ imamo $1 = 1$. Pretpostavimo da važi za $n - 1$.
    \[ \sum_{k = 0}^{n} {{k} \choose r} = \sum_{k = 0}^{n-1} {{k} \choose r} + {{n} \choose r} = {{n} \choose {r + 1}} + {{n} \choose r} = {{n + 1} \choose {r + 1}} \]
\end{proof}

\begin{theorem}
    \textbf{(Suma kvadrata):} Za $n \in \mathbb{N}_0$ važi
    \[ \sum_{k = 0}^{n} {{n} \choose k}^2 = {{2n} \choose n} \]
\end{theorem}
\begin{proof} Koeficijenti u z $x^n$ u narednim jednakostima moraju biti jednaki
    \[  \sum_{k = 0}^{2n}{{2n} \choose k} x^k = (x + 1)^{2n} = (x+1)^n (x+1)^n = \bigg(\sum_{k = 0}^{n}{{n} \choose k}x^k\bigg)\bigg(\sum_{k = 0}^{n}{{n} \choose k}x^{n-k}\bigg)  \]
    \[ \implies \sum_{k = 0}^{n} {{n} \choose k}^2 = {{2n} \choose n} \]
\end{proof}

\section{Princip uključenja-isključenja.}

\begin{theorem}
    Za konačne skupove $A_1, A_2, \ldots, A_n$ važi
    \[ \Bigg| \bigcup_{i = 1}^n A_i \Bigg| = \sum_{\emptyset \neq I \subset \mathbb{N}_n} (-1)^{|I| - 1} \Bigg| \bigcap_{i \in I} A_i \Bigg|\]
\end{theorem}

\begin{proof}
    \[x \in \bigcup_{i = 1}^n A_i \implies x \;\text{pripada nekim od } A_1, A_2, \ldots, A_n\]
Bez umanjenja opštosti, neka $x$ pripada skupovima $A_1, A_2, \ldots, A_k$. Tada važi
\[ x \in \bigcap_{i = 1}^k A_i \]
Dakle, $x$ doprinosi da se vrednost sa leve strane teoreme uveća za 1. Treba dokazati da to važi i za desnu stranu. Naime, doprinos elementa $x$ desnoj strani jednak je
\[ k - {{k} \choose 2} + {{k} \choose 3} - \ldots + (-1)^{k-1}{{k} \choose k} = {{k} \choose 0} - \sum_{i = 0}^{k}{{k} \choose i}(-1)^i = 1 - 0 = 1 \]
\end{proof}

\begin{itemize}

\subsection{Uređeni izbori sa ponavljanjem}
\begin{theorem}
    Broj preslikavanja $f: \mathbb{N}_k \rightarrow \mathbb{N}_n$ jednak je $n^k$.
\end{theorem}
\begin{proof}
    Indukcijom po $k$. Za $k = 1$ postoji $n$ preslikavanja. Pretpostavimo da tvrdnje važi za $k-1$.
    Dakle, $k - 1$ element skupa $\mathbb{N}_k$ se u $\mathbb{N}_n$ može preslikati na $n^{k - 1}$ način. Preostali element se može preslikati u bilo koji od $n$ elemenata iz $\mathbb{N}_n$. Dakle, ukupan broj je $n^{k - 1} \cdot n = n^k$.
\end{proof}
\begin{definition}
    Neka je $A$ proizvoljni podskup skupa $X$. \textbf{Karakteristična funkcija} skupa $A$, $\chi_A : X \rightarrow \{0, 1\}$, definisana je sa
    \[ \chi_A = \begin{cases}
        1,\;\; x \in A, \\
        0,\;\; x \not\in A.
    \end{cases} \]
\end{definition}

\begin{theorem}
    Neka je $X$ skup od $n$ elemenata. Tada važi
    \[ |P(X)| = 2^n \]
\end{theorem}
\begin{proof}
    Svakom podskupu skupa $X$ odgovara jedna karakteristična funkcija, a i svakoj karakterističoj funkciji odgovara jedan podskup. Ovih funkcija ima $2^n$.
\end{proof}

\subsection{Uređeni izbori bez ponavljanja}
\begin{itemize}
    \item Broj uređenih izbora $k$ elemenata skupa od $n$ elemenata jednak je broju injektivnih preslikavanja $f: \mathbb{N}_k \rightarrow \mathbb{N}_n$.
\end{itemize}
\begin{theorem}
    Za prirodne brojeve $n$ i $k$ td. $k \leq n$, broj injektivnih preslikavanja $f: \mathbb{N}_k \rightarrow \mathbb{N}_n$ jednak je
    \[ \prod_{i = 0}^{k - 1} (n-i)  = \frac{n!}{(n-k)!}\]
\end{theorem}

\begin{proof}
    Indukcijom po $k$. Za $k = 1$, imamo $n$ preslikavanja, slično kao ranije. Pretpostavimo da tvrdnje važi za $k - 1$. Sada prvih $k - 1$ elemenata skupa $\mathbb{N}_k$ možemo na $n(n-1)\dots(n - k + 2)$ načina preslikati u skup $\mathbb{N}_n$. Kako su u pitanju izbori bez ponavljanja, preostali element možemo pridružiti jednom od $n - k + 1$ elemenata. Dakle, ukupan broj načina je 
    \[ n(n-1)\dots (n - k + 1) \]
\end{proof}

\begin{itemize}
    \item \textbf{Permutacije} su specijalan slučaj u kom važi $n = k$. Dakle, broj permutacija je $n!$.
    \item Za veliko $n$, faktorijel se može aproksimirati \textbf{Stirlingovom formulom}: \[ n! \approx \sqrt{2\pi n}\bigg(\frac{n}{e}\bigg)^n \]
\end{itemize}

\begin{itemize}

\subsection{Neuređeni izbori bez ponavljanja}
\begin{theorem}
    Neka su $k, n \in \mathbb{N}, \; k \leq n$. Broj Neuređenih izbora $k$ elemenata skupa od $n$ elemenata je 
    \[ {{n} \choose k} = \frac{n!}{k!(n-k)!} \]
\end{theorem}
\begin{proof}
    Kako je broj uređenih izbora $k$ od $n$ elemenata jednak $\frac{n!}{(n-k)!}$, a nama trebaju neuređeni izbori, treba podeliti ovaj količnik sa $k!$.
\end{proof}
\newpage
\begin{theorem}
    \textbf{(Vandermondov identitet):}
    \[{{r}\choose {0}}{{s} \choose n} + {{r} \choose 1}{{s} \choose {n-1}} + {{r} \choose 2}{{s} \choose n - 2} + \dots + {{r} \choose r}{{s} \choose n - r} = {{r+s} \choose n}\]
\end{theorem}
\begin{proof}
    Neka se u prostoriji nalazi $r$ devojčica i $s$ dečaka. Neka je potrebno izabrati $n$ dece nezavisno od pola. To se može uraditi na ${{r+s}\choose n}$ načina. \\

    Sa druge strane, $k$ devojčica možemo izabrati na ${{r} \choose k}$ načina, dok $n - k$ dečaka možemo odabrati na ${{s} \choose {n-k}}$. Sumiranjem po $k$ dobijamo da je ukupan broj načina za odabir $n$ dece
    \[ \sum_{k = 0}^{r}{{r} \choose k}{{s} \choose {n-k}} \]
\end{proof}

\begin{theorem}
    \[ {{r} \choose 0}{{s} \choose n} + {{r} \choose 1}{{s} \choose n + 1} + {{r} \choose 2}{{s}\choose n + 2} + \dots + {{r} \choose r}{{s} \choose {{n + r}}} = {{r + s} \choose {r + n}} \]
\end{theorem}
\begin{proof}
    Prema Vandermondovom identitetu važi
    \[\sum_{k = 0}^{r}{{r} \choose k}{{s} \choose n + k} = \sum_{k = 0}^{r}{{r} \choose k}{{s} \choose {s - n - k}} = {{r + s} \choose {s - n}} = {{r+s} \choose {r+s-s+n}} = {{r+s} \choose {r+n}}\]
\end{proof}
\subsection{Neuređeni izbori sa ponavljanjem}

\begin{theorem}
    Broj neuređenih izbora $k$ elemenata skupa od $n$ elemenata uz dozvoljeno ponavljanje je
    \[ \frac{(n + k - 1)!}{k!(n-1)!} = {{n + k - 1} \choose k}\]
\end{theorem}
\begin{proof}
    Zamislimo tih $k$ elemenata kao $k$ identičnih kuglica. Želimo da tih $k$ kuglica rasporedimo u $n$ kutija. Ovih $n$ kutija određuje $n - 1$ pregrada. Dakle, za pregrade i kuglice ukupno imamo $k + n - 1$ mesta. Od njih biramo $k$ mesta za kuglice, što se može uraditi na ${{n + k -1} \choose k}$ načina.
\end{proof}


\begin{theorem}
    Broj permutacija sa ponavljanjem multiskupa od $n$ elemenata u kojem se $i$-ti element pojavljuje $n_i$ puta, $i \leq k$, td. $n = n_1 + n_2 + \dots + n_k$ jednak je
    \[ \frac{n!}{n_1! n_2! \dots n_k!} = {{n} \choose {n_1, n_2, \dots, n_k}} \]
\end{theorem}
\begin{proof}
    \[\frac{n!}{n_1!(n-n_1)!}\cdot \frac{(n-n_1)!}{n_2!(n - n_1 - n_2)!} \dots \frac{(n - n_1 - \dots - n_{k-1})!}{n_k!(n - n_1 - \dots -n_k)!} = \frac{n!}{n_1!\dots n_k!}\]
\end{proof}

\section{Generisanje permutacija i kombinacija}
\subsection{Generisanje permutacija}
\begin{itemize}
    \item Generisanje permutacija podrazumeva određivanje svih permutacija skupa u skladu sa nekim pravilom koje određuje redosled u kom se generišu.
    \item U \textbf{leksikografskom uređenju}, permutacija $A$ nalazi se ispred permutacije $B$ ako važi \[ a_1 = b_1, a_2 = b_2, \ldots, a_{k-1} = b_{k-1}, a_k < b_k \]
\end{itemize}
\vfill
\begin{algorithm}[!h]
    \caption{Naredna permutacija}
    \begin{itemize}
        \item \textbf{INPUT}: Permutacija $A = a_1a_2\dots a_n$
    \end{itemize}
    \begin{enumerate}
        \item \textbf{IF} Ako su brojevi $a_i, \;1 \leq i \leq n$ uređeni u opadajući niz
        \begin{itemize}
            \item onda je $A$ poslednja permutacija u leksikografskom uređenju, pa naredna ne postoji.
        \end{itemize} 
        \textbf{ELSE}
        \begin{itemize}
            \item U suprotnom, pređi na sledeći korak.
        \end{itemize}
        \item Pretražiti $A$ zdesna nalevo do prvog pojavljivanja indeksa $i$ za koji je $a_i < a_{i + 1}$.
        \item Odrediti $\min \{a_j \in A \;|\; a_j > a_i \;\land\; i < j\}$
        \item Zameniti mesta brojevima $a_i$ i $a_j$, potom urediti u rastući poredak sve brojeve koji se nakon zamene nalaze desno od $a_j$.
        \item \textbf{RETURN} $A$
    \end{enumerate}
\end{algorithm}
\vfill
\newpage
\begin{algorithm}[!h]
    \caption{$k$-ta permutacija}
    \begin{itemize}
        \item \textbf{INPUT} Skup $\mathbb{N}_n$
        \item $k^{(0)} = k$
        \item $a^p_1 = \bigg\lceil \frac{k^{(0)}}{(n-1)!}\bigg\rceil$
        \item $a_1 = a_1^p$
        \item \textbf{FOR} $i = 2$ \textbf{TO} $n$
        \begin{itemize}[label=$\circ$]
            \item $k^{(i-1)} = k^{(i-2)}-(a^p_{i-1}-1)(n-i+1)!$
            \item $a_i^p = \bigg\lceil \frac{k^{(i-1)}}{(n-i)!}\bigg\rceil$
            \item $a_i$ = [$a_i^p$-ti broj u rastućem poretku preostalih brojeva]
        \end{itemize}
        \item \textbf{RETURN} $A = a_1a_2\ldots a_n$
    \end{itemize}
\end{algorithm}
\begin{proof}
    Objasnimo kako ovaj algoritam radi. Ako fiksiramo prvi broj, ostale brojeve
    možemo rasporediti na $(n-1)!$ načina. Ovo možemo zamisliti kao $n$ blokova 
    od po $(n-1)!$ brojeva. Blokovi su organizovani po prvom broju - prvi blok
    sadrži sve permutacije koje počinju brojem $1$, drugi brojem $2$ itd. do $n$.
    Deljenjem trenutnog indeksa $k$ sa brojem permutacija u jednom bloku saznajemo
    u kom bloku se nalazi željena permutacija, tj. dobijamo prvi broj u permutaciji. Sada rekurzivno ponavljamo isti postupak,
    tj. delimo blokove na podblokove. Broj koji smo sada ubacili u permutaciju izbacujemo
    iz liste slobodnih brojeva, a novi indeks formiramo na sledeći način:
    $$k^{(i-1)} = \underbrace{k^{(i-2)}}_{\text{dosadašnji globalni indeks}} - \underbrace{(a_{i-1}^p - 1)}_{\text{broj blokova koje smo potpuno preskočili}} \cdot \underbrace{(n - i + 1)!}_{\text{veličina jednog bloka}}$$
    Dakle, da bismo resetovali brojač tako da opet krene od 1, moramo oduzeti broj 
    svih permutacija u blokovima koje smo preskočili. Kada uradimo ovaj postupak $n$ puta, pronašli smo celu permutaciju.
\end{proof}

\section{Rekurentne jednačine. Definicija i rešenja. Linearna rekurentna jednačina i
njeno opšte rešenje. }
\begin{definition}
    Za $n \in \mathbb{N}_0, k\in \mathbb{N}, a_i \in \mathbb{R}$, \textbf{rekurentnom jednačinom} nazivamo jednačinu oblika
    \[ a_{n+k} = \phi(n, a_n, a_{n+1}, \ldots, a_{n+k-1}).\]
    \textbf{Rešenje} rekurentne jednačine je bilo koji niz $(a_n)$ čiji članovi zadovoljavaju tu jednačinu. \textbf{Opšte rešenje} zavisi od $k$ konstanti, tj. članova niza. Različita rešenja se dobijaju iz opšteg izborima vrednosti konstanti. Uslove kojima se jednoznačno određuju vrednosti ovih konstanti zovemo \textbf{početnim uslovima}. Jedno rešenje koje je određeno nekim početnim uslovima zovemo \textbf{partukularno rešenje}. Za prozvoljnu vrednost $n$, $n$-ti član niza zovemo \textbf{opšti član}.
\end{definition}
\begin{itemize}
    \item Linearna rekurentna jednačina $k$-tog reda:
    \[
        f_0(n)a_n + f_1(n)a_{n+1} + \dots + f_k(n)a_{n + k} = f(n)
    \]
    \item Linearne rekurentne jednačine mogu imati konstantne ili funkcionalne koeficijente.
    \item Ako je $f(n) = 0$, onda je jednačina homogena.
    \item Ako je $f_k(n) = 1$, onda je jednačina u normalizovanom obliku.
    \item Rešenja $(a_n^{(1)}), \ldots, (a_n^{(r)})$ su zavisna ako se neko od njih može predstaviti kao lin. kombinacija ostalih.
\end{itemize}
\begin{theorem}
    Rešenja $(a_n^{(1)}), \ldots, (a_n^{(r)})$ homogene jednačine su nezavisna akko
    \[ \begin{vmatrix} 
        a_1^{(1)} & a_1^{(2)} & \ldots & a_1^{(r)} \\ 
        a_2^{(1)} & a_2^{(2)} & \ldots & a_2^{(r)} \\ 
        \vdots & \vdots & \ddots &\vdots \\ 
        a_r^{(1)} & a_r^{(2)} & \ldots & a_r^{(r)} 
    \end{vmatrix} \neq 0\]
\end{theorem}
\begin{proof}\
    \begin{itemize}
        \item $\implies$: \\
        Nezavisnost rešenja daje i nezavisnost kolona-vektora matrice, što znači da je determinanta različita od nule.
        \item $\impliedby$: \\
        Ako je determinanta različita od nule, tada je njen rang jednak $r$, pa su njeni kolona-vektori nezavisni, dakle rešenja su nezavisna.
    \end{itemize}
\end{proof}
\begin{theorem}
    Ukoliko su rešenja $(a_n^{(1)}), \ldots, (a_n^{(r)})$ homogene jednačine nezavisna, tada je opšte rešenje određeno sa
    \[ a_n = c_1a_n^{(1)} + c_2a_n^{(2)} + \ldots + c_ka_n^{(k)}, \;\; c_i \in \mathbb{R} \]
\end{theorem}
\begin{itemize}
    \item Opšte rešenje nehomogene linearne rekurentne jednačine je oblika 
    \[
        a_n = h_n + p_n,
    \]
    gde je $h_n$ opšte rešenje odgovarajuće homogene jednačine, a $p_n$ partikularno rešenje date nehomogene jednačine.
\end{itemize}

\section{Tipovi grafova, osnovni pojmovi, svojstva i invarijante. Šetnje, staze i putevi.
Matrične reprezentacije grafa.}
\subsection{Tipovi grafova}
\begin{itemize}
    \item \textbf{Prost graf}: $G = (V, E),\;\; V $ - skup čvorova, $E$ - skup grana.
    \item Čvorovi su \textit{susedni} ako ih grana spaja. Grana je \textit{incidentna} sa ova dva čvora. Ti čvorovi su \textit{krajevi} grane. Grane incidentne sa susednim čvorom su \textit{susedne}.
    \item \textbf{Stepen čvora} $u$ ($d_u$) - broj grana incidentnih sa njim.
    \item \textbf{Okolina čvora} $u$ ($N(u)$) - skup svih čvorova njemu susednih.
    \item \textbf{$\delta$} - \textit{minimalni stepen} čvora u grafu; \textbf{$\Delta$} - \textit{maksimalni stepen} čvora u grafu.
    \item \textbf{Graf sa petljama} - dopuštene grane čija su oba kraja isti čvor.
    \item \textbf{Multigraf} - dopušteno više grana između dva ista čvora.
    \item \textbf{Težinski graf} - svakoj grani pridružena numerička vrednost.
    \item \textbf{Usmereni (orijentisani) graf (digraf)} - svaka grana može imati usmerenje.
\end{itemize}

\begin{theorem}
    Suma stepena čvorova grafa jednaka je dvostrukom broju grana:
    $$\sum_{i = 1}^{n} d_{u_i} = 2m$$
\end{theorem}
\begin{proof}
    Prilikom sumiranja ovih sabiraka, svaku granu brojimo tačno 2 puta.
\end{proof}

\begin{theorem}
    Broj čvorova \textit{neparnog stepena} grafa nije neparan.
\end{theorem}
\begin{proof}
    Na osnovu prethodne teoreme, suma stepena svih čvorova je parna. Pošto isto važi i za sumu parnih stepena, suma neparnih stepena mora biti parna, pa je broj čvorova neparnog stepena paran.
\end{proof}

\subsection{Izomorfizmi grafova i teoreme}
\begin{itemize}
    \item \textbf{Izomorfni grafovi} - postoji \textit{bijekcija} skupa čvorova jednog na skup čvorova drugog koja čuva susedstva. U prevodu, izomorfni su ako se razlikuju samo oznake čvorova.
    \item \textbf{$k$-dimenziona kocka} - graf čiji se skup čvorova sastoji od svih uređenih binarnih $k$-torki i čiji su čvorovi susedni akko se odgovarajuće $k$-torke razlikuju tačno u jednoj koordinati.
    \item \textbf{Invarijanta grafa} - funkcija definisana na skupu grafova koja ima istu vrednost za sve izomorfne grafove.
    \item \textbf{Podgraf} $G' = (V', E')$ grafa $G = (V, E)$ - važi $V' \subseteq V, \; E' \subseteq E$.
    \item \textbf{Indukovani podgraf} - dobija se uklanjanjem proizvoljnog skupa čvorova grafa (zadržavaju se sve grane iz originalnog grafa koje povezuju preostale čvorove).
    \item \textbf{Komplement} $\overline{G} = (V, \overline{E})$ grafa $G = (V, E)$ - čvorovi su susedni u $\overline{G}$ akko nisu susedni u $G$. Graf je \textit{samokomplementaran} ako je izomorfan svom komplementu.
\end{itemize}
\subsection{Šetnje, staze i putevi}
\begin{itemize}
    \item \textbf{Šetnja} - niz čvorova i grana $(v_1, e_1, v_2, e_2, \ldots, e_{n-1}, v_n)$ grafa $G$ u kojem se svaka grana pojavljuje između dva sa njom incidentna čvora. \textit{Dužina šetnje} je broj grana koje se pojavljuju u njoj. Šetnja je \textit{zatvorena} akko počinje i završava u istom čvoru.
    \item \textbf{Staza} - šetnja u kojoj se svaka grana pojavljuje najviše jednom. 
    \item \textbf{Put} - šetnja u kojoj se svaki čvor (pa i svaka grana) pojavljuje najviše jednom. Put u usmerenom grafu određen je nizom različitih čvorova $(v_1, v_2, \ldots, v_n)$ takvim da taj usmereni graf sadrži grane $(v_i, v_{i+1})$, za $1 \leq i \leq k-1$.
    \item Dva čvora $u$ i $v$ su \textit{povezana} akko postoji šetnja koja počinje u $u$ i završava u $v$. Graf je \textit{povezan} akko su svi njegovi čvorovi međusobno povezani.
    \item Povezanost čvorova u grafu je relacija ekvivalencije. \textit{Komponente povezanosti} grafa su klase ekvivalencije te relacije.
    \item \textbf{Vezivni čvor} - čvor čije uklanjanje povećava broj komponenti povezanosti grafa.
    \item \textbf{Most} (ili vezivna grana) - grana čije uklanjanje povećava broj komponenti povezanosti grafa (za 1).
    \item \textbf{Rastojanje} $\operatorname{dist}(u, v)$ između dva čvora $u$ i $v$ - dužina najkraćeg puta koji ih povezuje. Ako ne postoji takva šetnja, rastojanje je beskonačno.
    \item \textbf{Ekscentricitet} čvora $u$ - $\operatorname{ecc}(u) = \max \{ \operatorname{dist}(u, v)$ | $v \in V \}$.
    \item \textbf{Radijus} grafa $G$ - $\operatorname{rad}(G) = \min \{ \operatorname{ecc}(u)$ | $u \in V \}$.
    \item \textbf{Centar} grafa $G$ - skup svih čvorova $u$ za koje važi $\operatorname{ecc}(u) = \operatorname{rad}(G)$.
    \item \textbf{Dijametar} grafa $G$ - $\operatorname{diam}(G) = \max \{ \operatorname{ecc}(u)$ | $u \in V \}$.
    \item Povezanost, radijus i dijametar grafa su invarijante.
\end{itemize}
\subsection{Neki grafovi}
\begin{itemize}
    \item \textbf{Prazan graf} - graf bez grana. Takav graf sa sa jednim čvorom se naziva \textit{trivijalni graf}.
    \item \textbf{Potpun (kompletan) graf} $K_n$ - graf sa $n$ čvorova od kojih su svaka dva povezana granom.
    \item \textbf{Regularni graf} - graf u kojem su svi čvorovi istog stepena. Ako je stepen $k$, graf se naziva \textit{$k$-regularnim}.
    \item \textbf{Bipartitni graf} - trivijalni graf ili graf čiji se skup čvorova može podeliti na dva disjunktna podskupa tako da svaka grana povezuje čvor iz jednog podskupa sa čvorom iz drugog.
    \newpage
    \item \textbf{Potpun (kompletan) bipartitni graf} $K_{m, n}$ - bipartitni graf sa $m$ čvorova u jednom i $n$ čvorova u drugom podskupu, pri čemu je svaki čvor iz jednog podskupa povezan sa svakim čvorom iz drugog podskupa. Stepen svakog čvora iz prvog podskupa je $n$, a stepen svakog čvora iz drugog podskupa je $m$.
    \begin{itemize}
        \item \textbf{Zvezda} $S_n$ - kompletan bipartitni graf $K_{1, n}$.
    \end{itemize}
    \item \textbf{Ciklus} $C_n$ - Povezan graf kod kojeg je stepen svakog čvora jednak 2. Ciklus sa $n$ čvorova se naziva \textit{$n$-ciklus}.
    \item \textbf{Put} je:
    \begin{itemize}
        \item za $n \leq 2$, jedini povezan graf sa $n$ čvorova;
        \item za $n \geq 3$, graf koji se dobija uklanjanjem jedne grane ciklusa $C_n$.
    \end{itemize} Put koji ima $n$ čvorova označava se sa $P_n$.
    \item  \textbf{Dužina ciklusa} i \textbf{dužina puta} - broj grana koje sadrže.
    \item Put u grafu određuje indukovani podgraf koji je put.
\end{itemize}
\begin{theorem}
    Graf je bipartitan. $\iff$ Graf ne sadrži cikluse neparne dužine.
\end{theorem}
\begin{proof}\
    \begin{itemize}
        \item $\implies$: 
        Ako bi bipartitni graf sadržao ciklus neparne dužine, tada bi svaka dva susedna čvora tog ciklusa pripadala različitim skupovima particije čvorova, međutim to nije moguće, jer ciklus sadrži neparan broj čvorova.
        \item $\impliedby$:
        Vršimo razvrstavanje čvorova grafa u dva skupa td. čvorovi koji pripadaju istom skupu nisu susedni. Najpre proizvoljan čvor, npr. $u$, dodajemo u prvi skup. Zatim sve čvorove koji su na neparnom rastojanju od $u$ dodajemo u drugi skup, a sve one koji su na parnom dodajemo u prvi. Ako, nakon završetka ovog postupka, među čvorovima istog skupa nema susednih, graf je bipartitan. \\ U suprotnom, dogodiće se da su ova dva čvora, koja su ili oba na parnom ili oba na neparnom rastojanju od $u$, susedni. U tom slučaju, kombinovanjem najkraćih puteva od $u$ do svakog od tih čvorova i grane koja ih spaja, dobijamo ciklus neparne dužine, što je u suprotnosti sa pretpostavkom. \\ Ako je graf nepovezan, onda prethodnu proceduru treba primeniti na svaku od komponenata povezanosti.
    \end{itemize}
\end{proof}
\subsection{Matrične reprezentacije grafova}
\begin{itemize}
    \item \textbf{Matrica susedstva} $A(G)$ grafa $G$ sa $n$ čvorova - kvadratna matrica reda $n$ čiji element $a_{ij}$ je jednak broju grana koje povezuju čvorove $v_i$ i $v_j$. Za prost graf, $a_{ij} = 1$ ako su čvorovi susedni, a $a_{ij} = 0$ inače.
    \item \textbf{Matrica incidencije} $R(G)$ grafa $G$ sa $n$ čvorova i $m$ grana - matrica reda $n \times m$ čiji element $a_{ij}$ je jednak 1 ako je čvor $v_i$ incidentan sa granom $e_j$, a $a_{ij} = 0$ inače.
\end{itemize}

\section{Stabla. Korenska i razapinjuća stabla. Pretrage grafova. }
\subsection{Definicija i osnovna svojstva}
\begin{itemize}
    \item \textbf{Stablo} - Povezan graf koji ne sadrži podgraf izomorfan ciklusu. U prevodu, povezan graf bez ciklusa.
    \item \textbf{Šuma} - Graf čije su komponente povezanosti stabla.
\end{itemize}
\begin{theorem}
    Sledeće tvrdnje su ekvivalentne:
    \begin{enumerate}
        \item $T$ je stablo.
        \item $T$ je povezan graf kod kojeg je broj grana za 1 manji od broja čvorova, tj važi $m = n - 1$.
        \item $T$ je maksimalni graf bez ciklusa, tj. dodavanje bilo koje grane u $T$ stvara ciklus.
        \item $T$ je minimalni povezan graf, tj. uklanjanje bilo koje grane iz $T$ ga čini nepovezanim.
        \item Za svaka dva čvora grafa $T$ postoji tačno jedan put koji ih povezuje.
    \end{enumerate}
\end{theorem}
\subsection{Korenska i razapinjuća stabla}
\begin{itemize}
    \item \textbf{Korensko stablo} - stablo u čiji je jedan čvor izdvojen od ostalih. Taj čvor se naziva \textit{koren}.
    \item Ako je $r$ koren stabla i ako važi $\operatorname{dist}(u, r) = i$, kažemo da čvor $u$ pripada \textit{$i$-tom nivou} stabla. Ako $u$ pripada $i$-tom nivou, a $v$ $(i + 1)$-om nivou, i ako su susedni, kažemo da je $u$ \textit{prethodnik} čvora $v$, a $v$ \textit{sledbenik} čvora $u$. Čvor koji nema naslednika nazivamo \textit{list}. Čvor koji nije ni koren ni list nazivamo \textit{unutrašnji čvor}.
    \item \textbf{Korensko $k$-arno stablo} - korensko stablo kod kojeg svaki čvor ima najviše $k$ sledbenika.
    \item \textbf{Potpuno korensko $k$-arno stablo} - korensko $k$-arno stablo kod kojeg svaki čvor koji nije list ima tačno $k$ sledbenika.
    \item \textbf{Razapinjući graf} grafa $G$ - graf koji ima isti skup čvorova kao i $G$ i čiji je skup grana podskup skupa grana grafa $G$. 
    \item \textbf{Razapinjuće stablo} povezanog grafa $G$ - razapinjući graf koji je stablo.
    \item \textbf{Razapinjuća šuma} grafa $G$ - razapinjući podgraf koji se sastoji od razapinjućih stabala komponenata povezanosti grafa $G$.
\end{itemize}

\subsection{Pretrage grafova}
Pretraga grafa je postupak sistematskog obilaska svakog čvora grafa tačno jednom. Rezultat pretrage je \textbf{stablo pretrage} (razapinjuće usmereno stablo).

\begin{itemize}
    \item \textbf{Pretraga po dubini} (DFS - Depth First Search) - „najdublji” mogući put: iz trenutnog čvora se ide ka neobrađenom susedu dokle god je moguće, a zatim se vrši povratak (bektrek) na prethodni čvor. 
    \item \textbf{Pretraga po širini} (BFS - Breadth First Search) - „nivo po nivo”: prvo se obilaze svi neobrađeni susedi početnog čvora, zatim svi neobrađeni susedi tih suseda, i tako redom.
\end{itemize}

\section{Minimalno razapinjuće stablo težinskog grafa. Kraskalov i Primov algoritam. }

\subsection{Težinski grafovi}
Kod težinskih grafova, svakoj grani $e$ dodeljena je vrednost (težina) $w(e)$, koja predstavlja trošak, udaljenost ili kapacitet veze.

\begin{itemize}
    \item \textbf{Težina grafa} $w(G)$ - suma težina svih grana u grafu:
    $$w(G) = \sum_{e \in E(G)} w(e)$$
    \item \textbf{Minimalno razapinjuće stablo} (MST) - razapinjuće stablo $T$ grafa $G$ za koje je suma težina grana $w(T)$ minimalna.
\end{itemize}

\subsection{Kraskalov algoritam}
Strategija Kraskalovog algoritma je „najpre grana najmanje težine”. Reč je o \textbf{pohlepnom algoritmu} koji u svakom koraku bira lokalno optimalno rešenje.
\begin{algorithm}[ht]
    \caption{Kraskalov algoritam}
    \begin{enumerate}
        \item \textbf{INPUT}: Povezan težinski graf $G = (V, E)$.
        \item $T = (V, \emptyset)$ // inicijalizacija praznog stabla.
        \item Sortiraj sve grane $e \in E$ u neopadajući niz po težinama.
        \item \textbf{FOR} svaku granu $e = (u, v)$ iz sortiranog niza:
        \begin{itemize}
            \item \textbf{IF} $T + e$ ne sadrži ciklus:
            \item $T := T + e$ // dodaj granu u stablo.
        \end{itemize}
        \item \textbf{RETURN} $T$
    \end{enumerate}
\end{algorithm}

\begin{theorem}
    Kraskalov algoritam rezultuje minimalnim razapinjućim stablom (MST) povezanog težinskog grafa.
\end{theorem}
\begin{proof}
    Najpre, algoritam ne može rezultovati nepovezanim grafom, niti grafom koji sadrži ciklus (videti korak K4), što znači da takav graf mora biti stablo.

    Neka je nadalje $T$ stablo dobijeno Kraskalovim algoritmom i neka su $e_1, e_2, \dots, e_{n-1}$ grane tog stabla uređene u skladu sa redosledom dodavanja tekućem grafu. Pretpostavimo suprotno, to jest da težina stabla $T$ nije minimalna. U tom slučaju, postoje razapinjuća stabla (najmanje jedno) minimalne težine koja je manja od težine stabla $T$. Neka je za svako takvo stablo $S$ definisana funkcija $f(S) = \min\{i \mid e_i \notin E(S)\}$.

    Označimo sa $S'$ stablo za koje je vrednost ove funkcije maksimalna. Ako važi $f(S') = k$, onda su grane $e_1, e_2, \dots, e_{k-1}$ istovremeno i grane stabla $S'$. U skladu sa Teoremom 3.2.1 (deo (1)$\implies$(3)), graf koji se dobija dodavanjem grane $e_k$ stablu $S'$ (dakle, $S' + e_k$) sadrži ciklus. Ako je $e'_k$ grana tog ciklusa koja pripada stablu $S'$, a ne pripada $T$ (takva grana postoji jer se stabla $T$ i $S'$ razlikuju), razmotrimo graf $S''$ koji se dobija uklanjanjem grane $e'_k$ iz $S'$ i potom dodavanjem grane $e_k$. Broj grana grafa $S''$ takođe je jednak $n - 1$, a graf je uz to i povezan (jer je grana $e'_k$ pripadala ciklusu), dakle i on je (razapinjuće) stablo.

    Jasno je da važi jednakost
    \begin{equation}
        w(S'') = w(S') + w(e_k) - w(e'_k).
        \label{Kraskal1}
    \end{equation}

    Primetimo sada da stabla $T$ i $S''$ sadrže grane $e_1, e_2, \dots, e_k$. Pošto je grana $e_k$ sledeća koja je u skladu sa algoritmom dodata stablu $T$, mora važiti $w(e_k) \le w(e'_k)$, što zajedno sa jednakošću (\ref{Kraskal1}) daje nejednakost $w(S'') \le w(S')$. Zapravo važi jednakost, jer je $S'$ stablo minimalne težine. Međutim, važi i nejednakost $f(S'') > k = f(S')$, što protivureči izboru stabla $S'$.
\end{proof}

\begin{itemize}
    \item \textbf{Napomena:} Algoritam se može optimizovati uvođenjem brojača dodatih grana. Kada broj grana dostigne $n-1$, algoritam se može završiti, jer je stablo kompletirano.
\end{itemize}


\subsection{Primov algoritam}
Strategija Primovog algoritma zasniva se na principu „najpre najbliži sused”. Za razliku od Kraskalovog algoritma koji grane bira globalno, Primov algoritam gradi stablo postepeno, proširujući trenutni podgraf.

\begin{theorem}
    Neka je $G = (V, E)$ povezan težinski graf, $U \subset V$ i $e$ grana minimalne težine koja spaja čvor iz skupa $U$ sa čvorom iz skupa $V \setminus U$. Tada postoji minimalno razapinjuće stablo grafa $G$ koje sadrži granu $e$.
\end{theorem}
\begin{proof}
    Neka je $S$ minimalno razapinjuće stablo koje ne sadrži granu $e$. Podgraf $S + e$ sadrži ciklus. Pored grane $e$, tom ciklusu pripada i neka grana $e'$ koja spaja čvor iz skupa $U$ sa čvorom iz skupa $V \setminus U$. Pošto važi $w(e) \le w(e')$, uklanjanjem grane $e'$ iz grafa $S + e$ dobijamo (traženo) minimalno razapinjuće stablo.
\end{proof}

\begin{algorithm}[ht]
    \caption{Primov algoritam}
    \begin{enumerate}
        \item \textbf{INPUT}: Težinski graf $G = (V, E)$.
        \item $T = (\emptyset, \emptyset)$ // inicijalizacija praznog stabla.
        \item Izaberi proizvoljan čvor $v \in V$ i dodaj ga u $T$.
        \item \textbf{WHILE} broj čvorova u $T < |V|$:
        \begin{itemize}
            \item Nađi granu $e = (u, w)$ minimalne težine takvu da $u \in T$, a $w \notin T$.
            \item Dodaj granu $e$ i čvor $w$ u $T$.
        \end{itemize}
        \item \textbf{RETURN} $T$
    \end{enumerate}
\end{algorithm}

\section{Razapinjuća stabla označenih grafova. Priferov niz}

\begin{definition}
    Za $n \geq 2$, \textbf{Priferov niz}, $s_{(n)} = (s_1, s_2, \ldots, s_{n-2})$, je bilo koji niz koji se sastoji od $n-2$ (ne nužno različita) prirodna broja skupa $\mathbb{N}_n$.
\end{definition}

\begin{algorithm}[!h]
    \caption{Priferovo kodiranje}
    \begin{enumerate}
        \item \textbf{INPUT}: Stablo $T$ sa $n$ čvorova.
        \item \textbf{WHILE} broj čvorova $> 2$:
        \begin{itemize}
            \item $u$ = list sa najmanjom oznakom;
            \item $v$ = sused čvora $u$;
            \item $s := s + v$; // dodaj $v$ u niz
            \item $T := T - u$; // ukloni $u$ iz stabla
        \end{itemize}
        \item \textbf{RETURN} $s$
    \end{enumerate}
\end{algorithm}
\begin{algorithm}[!h]
    \caption{Priferovo dekodiranje}
    \begin{enumerate}
        \item \textbf{INPUT}: Priferov niz $s = (s_1, s_2, \ldots, s_{n-2})$.
        \item \textbf{Inicijalizacija}:
        \begin{itemize}
            \item $T = (V, \emptyset)$ gde je $V = \{1, 2, \ldots, n\}$.
            \item $l = (1, 2, \ldots, n)$ // lista čvorova koji mogu biti listovi.
        \end{itemize}
        \item \textbf{WHILE} $s \neq ()$:
        \begin{itemize}
            \item $v$ = najmanji čvor iz $l$ koji se ne nalazi u $s$;
            \item $w = s_1$;
            \item $T := T + \{v, w\}$ // dodaj granu između $v$ i $w$;
            \item $l := l \setminus \{v\}$;
            \item $s := s \setminus \{s_1\}$;
        \end{itemize}
        \item \textbf{Završni korak}: Poveži granom poslednja dva čvora preostala u listi $l$.
        \item \textbf{RETURN} $T$
    \end{enumerate}
\end{algorithm}

\section{Broj razapinjućih stabala označenih grafova. Kejlijeva teorema. Ciklomatički broj grafa.}
\subsection{Prebrojavanje označenih stabala}
\begin{theorem}[Kejlijeva teorema]
    Broj označenih stabala sa $n$ čvorova jednak je $n^{n-2}$.
\end{theorem}

\begin{itemize}
    \item \textbf{Kontrakcija} $G \cdot e$ - dobija se od multigrafa $G-e$ identifikovanjem (spajanjem) krajnjih čvorova grane $e$ u jedan čvor.
\end{itemize}
\begin{theorem}
    Za svaki označeni multigraf $G$ važi jednakost
    \[ t(G) = t(G - e) + t(G \cdot e), \]
    pri čemu se multigraf $G - e$ dobija uklanjanjem grane $e$, a multigraf $G \cdot e$ kontrakcijom multigrafa $G$ u odnosu na granu $e$.
\end{theorem}

\begin{proof}
    Svako razapinjuće stablo multigrafa $G$ koje ne sadrži granu $e$ jednoznačno određuje razapinjuće stablo multigrafa $G - e$ i obratno, što daje prvi sabirak sa desne strane jednakosti iz formulacije teoreme. Slično, svako razapinjuće stablo multigrafa $G$ koje sadrži granu $e$ jednoznačno određuje razapinjuće stablo multigrafa $G \cdot e$ i obratno, što daje drugi sabirak.
\end{proof}
\begin{itemize}
    \item Ako je $T$ razapinjuće stablo povezanog grafa $G$, onda se dodavanjem proizvoljne grane stablu $T$, pri čemu ona pripada grafu $G$, dobija tačno jedan ciklus u novonastalom grafu. Taj ciklus se naziva \textbf{elementarni ciklus} grafa $G$ u odnosu na razapinjuće stablo $T$.
    \item Broj elementarnih ciklusa povezanog grafa ne zavisi od izbora razapinjućeg stabla. Broj grana svakog razapinjućeg stabla jednak je $n - 1$, a po definiciji svaka grana koja pripada grafu, a ne pripada fiksiranom razapinjućem stablu, određuje jedan elementarni ciklus. Dakle, broj elementarnih ciklusa povezanog grafa $G$ jednak je $m - n + 1$.
\end{itemize}

\begin{theorem}
    Broj elementarnih ciklusa grafa jednak je $m - n + p$, gde je $m$ broj grana grafa, $n$ je broj čvorova grafa, a $p$ je broj komponenti povezanosti grafa.
\end{theorem}
\begin{proof}
    Indukcijom po broju komponenata povezanosti.
\end{proof}
\begin{itemize}
    \item \textbf{Ciklomatički broj} $v$ grafa - broj elementarnih ciklusa grafa.
\end{itemize}

\section{Ojlerovi multigrafovi. Flerijev algoritam. Problem kineskog poštara. }



Multigraf je \textbf{Ojlerov} ako sadrži zatvorenu stazu koja svaku granu obilazi tačno jednom. \textbf{Poluojlerov} graf sadrži stazu (ne nužno zatvorenu) koja svaku granu obilazi tačno jednom.

\begin{theorem}[Ojlerova teorema]
    Netrivijalan povezan multigraf je Ojlerov akko svi njegovi čvorovi imaju paran stepen.
\end{theorem}
\begin{proof}
    Označimo multigraf iz formulacije teoreme sa $G$ i pretpostavimo najpre da $G$ sadrži Ojlerovu (zatvorenu) stazu. Ukoliko se krećemo duž takve staze, tada kada god nekom granom dođemo do nekog čvora, moramo koristiti neku drugu granu (koju još nismo koristili) da odemo od tog čvora. Pošto ovakva staza prolazi svim granama uz završetak u polaznom čvoru, zaključujemo da su stepeni svih čvorova parni.

    Pretpostavimo sada da je stepen svakog čvora multigrafa $G$ paran. Dokaz ćemo izvesti metodom matematičke indukcije po broju grana. Minimalni broj grana netrivijalnog multigrafa kod kojeg je stepen svakog čvora paran je 2. Takav multigraf ima 2 čvora i 2 grane koje ih spajaju i za njega tvrđenje očigledno važi.

    Pretpostavimo da tvrđenje važi za sve multigrafe koji imaju manje od $m$ grana i da $G$ ima $m$ grana. Budući da su svi čvorovi parnog stepena, $G$ ne sadrži nijedan čvor stepena 1, pa prema tome $G$ nije stablo (što nije teško zaključiti, a čitalac se može setiti i zadatka 2 iz Odeljka 3.2). U tom slučaju, $G$ sadrži ciklus, recimo $C$. (U slučaju multigrafova ciklus može imati i 2 čvora; takav je na primer ciklus iz induktivne baze.) Ako uklonimo sve grane koje pripadaju ciklusu $C$, dobićemo multigraf koji ne mora biti povezan, ali koji i dalje ne sadrži čvorove neparnog stepena.

    Pretpostavimo da novonastali multigraf ima $k$ komponenata povezanosti i označimo ih sa $G_1, G_2, \dots, G_k$. Svaka od komponenata povezanosti $G_i$, u skladu sa induktivnom hipotezom, sadrži Ojlerovu stazu, recimo $s_i$.

    Pošto je multigraf $G$ povezan, svaka od staza $s_1, s_2, \dots, s_k$ ima barem jedan zajednički čvor sa ciklusom $C$. Traženu Ojlerovu stazu dobijamo tako što krećući se duž ciklusa $C$ kada dođemo do prvog čvora, recimo $u$, koji pripada zatvorenoj stazi $s_i$ koju nismo obišli, izađemo iz ciklusa, obiđemo celu stazu $s_i$ i vratimo se u čvor $u$. Kretanje završavamo kada dođemo do čvora ciklusa $C$ od kojeg smo krenuli.
\end{proof}

\begin{theorem}
    Netrivijalan povezan multigraf je poluojlerov akko sadrži tačno 0 ili 2 čvora neparnog stepena.
\end{theorem}
\begin{proof}
    Direktno na osnovu definicije poluojlerovog multigrafa i Ojlerove teoreme.
\end{proof}

Flerijev algoritam se koristi za pronalaženje Ojlerove staze u grafu. Ključno pravilo pri izboru grane je izbegavanje „mostova” osim ako ne postoji druga mogućnost.
\newpage
\begin{algorithm}[!h]
    \caption{Flerijev algoritam}
    \begin{enumerate}
        \item \textbf{INPUT}: Ojlerov multigraf $G$.
        \item Izaberi proizvoljan čvor $v_0$. Neka je $s = (v_0)$ trenutna staza.
        \item $H = G$ // radna kopija grafa.
        \item \textbf{WHILE} broj grana u $H > 0$:
        \begin{itemize}
            \item Izaberi granu $e = (u, v)$ iz $H$ takvu da:
            \begin{enumerate}
                \item $u$ je poslednji čvor staze $s$.
                \item $e$ nije most u $H$ (osim ako ne postoji drugi izbor).
            \end{enumerate}
            \item $s := s + e + v$; 
            \item $H := H - e$; // ukloni granu iz pomoćnog grafa.
        \end{itemize}
        \item \textbf{RETURN} $s$
    \end{enumerate}
\end{algorithm}



U slučaju kada graf nije poluojlerov, problem se može modifikovati dodavanjem novih grana ili dozvoljavanjem višestrukih prolazaka kroz iste grane.


\subsection{Problem kineskog poštara}
Problem kineskog poštara (engl. \textit{Route Inspection Problem}) odnosi se na pronalaženje zatvorene šetnje u povezanom težinskom multigrafu koja sadrži sve grane grafa i ima minimalnu ukupnu težinu.

\begin{itemize}
    \item \textbf{Cilj:} Optimizovati rutu tako da se obiđu sve ulice (grane) uz minimalan utrošak vremena (dužine puta).
    \item \textbf{Ojlerov slučaj:} Ako je multigraf Ojlerov, svaka Ojlerova staza predstavlja optimalno rešenje jer se svakom granom prolazi tačno jednom.
    \item \textbf{Opšti slučaj:} Ako graf nije Ojlerov (postoje čvorovi neparnog stepena), neke grane se moraju obići više puta. Problem se tada svodi na „uparivanje” čvorova neparnog stepena i dodavanje puteva minimalne težine između njih kako bi svi čvorovi postali parnog stepena.
\end{itemize}



\begin{algorithm}[!h]
    \caption{Rešavanje problema kineskog poštara (za neojlerov graf)}
    \begin{enumerate}
        \item Identifikuj sve čvorove neparnog stepena u multigrafu.
        \item Odredi sve moguće puteve između parova čvorova neparnog stepena i pronađi onaj skup uparivanja koji daje minimalnu ukupnu težinu.
        \item Dodaj „virtuelne” grane (duplikate postojećih grana) duž najkraćih puteva između odabranih parova, čineći graf Ojlerovim.
        \item Primenom Flerijevog algoritma na tako modifikovan graf pronađi Ojlerovu stazu.
    \end{enumerate}
\end{algorithm}


\section{Hamiltonovi grafovi. Problem trgovačkog putnika. 3-heuristika. }
Graf je \textbf{Hamiltonov} ako sadrži ciklus koji prolazi kroz svaki čvor grafa tačno jednom (\textbf{Hamiltonov ciklus}). Graf je \textbf{polu-Hamiltonov} ako sadrži put koji prolazi kroz svaki čvor grafa tačno jednom (\textbf{Hamiltonov put}).

\begin{theorem}[Dirakova teorema]
    Ako graf $G$ ima $n \geq 3$ čvorova i minimalni stepen čvora $\delta(G) \geq \frac{n}{2}$, tada je $G$ Hamiltonov.
\end{theorem}



\subsection{Problem konjičkog skoka}
Problem konjičkog skoka zahteva obilazak svakog polja šahovske table (ili table formata $m \times n$) skakačem, tako da se na svakom polju stane tačno jedanput.

\begin{itemize}
    \item \textbf{Grafovska interpretacija:} Čvorovi grafa su polja table, a grane predstavljaju dozvoljene poteze skakača.
    \item \textbf{Veza sa teorijom:} Postojanje rešenja ekvivalentno je postojanju Hamiltonovog puta (ili ciklusa) u tako konstruisanom grafu.
    \item \textbf{Napomena:} Za table formata $m \times n$ (gde je $m, n \geq 3$), rešenje postoji osim za tabele formata $3 \times 3$, $3 \times 5$, $3 \times 6$ i $4 \times 4$.
    \item \textbf{Metoda:} Problemi većih dimenzija (npr. $8 \times 8$) često se rešavaju metodom „podela i vladaj” — deljenjem table na manje segmente koji se pojedinačno rešavaju i povezuju.
\end{itemize}



Za razliku od Ojlerovih grafova, ne postoji jednostavna i efikasna teorema (nužan i dovoljan uslov) za proveru postojanja Hamiltonovog ciklusa u opštem grafu, zbog čega je ovaj problem računski složen (NP-kompletan).

\subsection{Problem trgovačkog putnika}
Problem trgovačkog putnika (engl. \textit{Traveling Salesperson Problem - TSP}) zahteva pronalaženje Hamiltonovog ciklusa minimalne težine u potpunom težinskom grafu.

\begin{itemize}
    \item \textbf{Cilj:} Obići svaki grad (čvor) tačno jednom i vratiti se u početni, uz minimalnu ukupnu dužinu puta.
    \item \textbf{Kompleksnost:} Problem je \textbf{NP-težak}, što znači da ne postoji poznat algoritam koji bi ga rešio u polinomijalnom vremenu za sve ulaze.
    \item \textbf{Pristupi:}
    \begin{itemize}
        \item \textit{Egzaktni algoritmi:} Pretraga svih mogućih Hamiltonovih ciklusa (neefikasno za veliki broj gradova).
        \item \textit{Heuristički algoritmi:} Traže \textbf{suboptimalna} rešenja (rešenja "bliska" optimalnom) u razumnom vremenu.
    \end{itemize}
\end{itemize}

\subsection{3-heuristika}
Ova heuristika poboljšava početni Hamiltonov ciklus kroz iterativne promene (lokalna pretraga).

\begin{algorithm}[!h]
    \caption{3-heuristika za TSP}
    \begin{enumerate}
        \item \textbf{INPUT}: Hamiltonov ciklus $s_1$ potpunog težinskog grafa.
        \item \textbf{WHILE} kriterijum zaustavljanja nije ispunjen:
        \begin{itemize}
            \item Ukloni 3 nesusedne grane iz trenutnog ciklusa $s_i$, čime se dobijaju 3 zasebna puta.
            \item Ponovo poveži krajeve tih puteva na sve moguće načine da se formira novi Hamiltonov ciklus.
            \item Izaberi ciklus $s_{i+1}$ sa minimalnom težinom među svim novonastalim ciklusima.
            \item \textbf{IF} $w(s_{i+1}) < w(s_i)$, prihvati $s_{i+1}$ kao trenutni ciklus.
        \end{itemize}
        \item \textbf{RETURN} $s_i$ (suboptimalni ciklus)
    \end{enumerate}
\end{algorithm}

\begin{itemize}
    \item \textbf{Napomena:} Strategija „uvek idi do najbližeg suseda” (Greedy) je najjednostavnija heuristika, ali često ne daje optimalno rešenje. 3-heuristika daje znatno bolje rezultate jer omogućava „preslaganje” ciklusa.
\end{itemize}

\section{Planarni grafovi.}

Graf je \textbf{planaran} ako se može nacrtati u ravni tako da se njegove grane nigde ne seku (osim u čvorovima). Svaki takav crtež deli ravan na jednu ili više \textbf{oblasti} (jedna od njih je uvek beskonačna).
\begin{theorem}[Ojlerova teorema za planarne grafove]
    Za svaki povezan planaran graf $G$ važi:
    $$n - m + f = 2$$
    gde su $n$ broj čvorova, $m$ broj grana, a $f$ broj oblasti na koje graf deli ravan.
\end{theorem}
\begin{proof}
    Indukcijom po broju grana. Pošto je $G$ povezan, minimalni broj grana koje sadrži jednak je $n - 1$ i tada je $G$ stablo. U slučaju stabla očigledno važi $f = 1$, odakle sledi tvrđenje. Pretpostavimo da tvrđenje važi za sve povezane grafove sa manje od $m$ grana i dokažimo da važi za graf $G$ koji ima $m \ge n$ grana. Pošto naš graf ima najmanje $n$ grana, on sadrži najmanje jedan ciklus. Neka je $e$ grana tog ciklusa. Graf $G - e$ je povezan (jer $e$ pripada ciklusu) i deli ravan na jednu oblast manje od grafa $G$ (jer $e$ ograničava tačno dve oblasti). Prema induktivnoj hipotezi (primenenoj na graf $G - e$) važi $n - (m - 1) + (f - 1) = 2$, odakle sledi tražena jednakost.
\end{proof}


\begin{itemize}
    \item \textbf{Potpodela grafa} dobija se zamenom svake grane tog grafa netrivijalnim putem proizvoljne dužine (koji spaja čvorove koji su bili incidentni sa granom). Dakle, potpodela se dobija uzastopnim dodavanjem proizvoljnog broja čvorova svakoj grani.
\end{itemize}
\begin{theorem}
    Graf je planaran ako i samo ako ne sadrži podgraf koji je izomorfan nekoj potpodeli grafa $K_5$ ili $K_{3,3}$.
\end{theorem}

\begin{itemize}
    \item \textbf{Značaj:} Ovo je fundamentalni rezultat koji nam omogućava da testiramo planarnost bilo kog grafa identifikacijom tzv. "{zabranjenih}" \;{struktura} (Kuratovskijevi grafovi).
\end{itemize}

\section{Sparivanja u grafovima.}
\textbf{Sparivanje} u grafu $G$ je podskup skupa grana takav da nijedne dve grane nemaju zajednički čvor (nisu susedne).

\begin{itemize}
    \item \textbf{Maksimalno sparivanje} - sparivanje kojem se ne može dodati nova grana, a da ostane sparivanje.
    \item \textbf{Najveće sparivanje} - sparivanje sa najvećim mogućim brojem grana.
    \item \textbf{Savršeno sparivanje} - sparivanje u kojem je svaki čvor grafa incidentan sa tačno jednom granom sparivanja. Ako graf ima neparan broj čvorova, onda ne može imati savršeno sparivanje.
\end{itemize}

Neka je $M$ proizvoljno sparivanje u grafu $G$. Za čvor $u$ tog grafa koji nije incidentan ni sa jednom granom sparivanja $M$ kažemo da je \emph{slobodan} u odnosu na $M$. Za put
\[ P = (v_1, e_1, v_2, e_2, \dots, e_{k-1}, v_k) \]
u grafu $G$ kažemo da je \emph{$M$-alternirajući} ukoliko ili sve grane neparnog indeksa ili sve grane parnog indeksa puta $P$ pripadaju sparivanju $M$. Drugim rečima, grane puta naizmenično pripadaju i ne pripadaju datom sparivanju. \emph{$M$-uvećani put} je $M$-alternirajući put kod kojeg su samo krajnji čvorovi slobodni u odnosu na sparivanje $M$. Takav put ne mora da postoji.


\begin{theorem}[Beržova teorema]
    Sparivanje $M$ u grafu $G$ je najveće ako i samo ako $G$ ne sadrži M-uvećani put.
\end{theorem}
\begin{proof}
    Pretpostavimo da je sparivanje $M$ najveće i neka, nasuprot tvrđenju, graf $G$ sadrži $M$-uvećani put, recimo $P$. Primetimo da je $M \Delta E(P)$ (pri čemu oznake $\Delta$ i $E(P)$ imaju isto značenje kao u Primeru 3.8.3) takođe sparivanje u razmatranom grafu. Štaviše, to sparivanje sadrži jednu granu više (od $M$), što znači da $M$ nije najveće.

    Drugu implikaciju dokazaćemo tako što ćemo dokazati kontrapoziciju, to jest pretpostavićemo da sparivanje $M$ nije najveće i dokazaćemo da tada graf sadrži $M$-uvećani put. Ako $M$ nije najveće sparivanje, onda postoji sparivanje $M'$ koje je najveće i time sadrži više grana od $M$. Razmotrimo podgraf grafa $G$ čiji je skup grana određen unijom skupova $M$ i $M'$. Stepen svakog čvora takvog podgrafa nije veći od 2. (Svaki čvor incidentan je sa najviše dve grane koje pripadaju različitim sparivanjima.) Drugim rečima, sve komponente povezanosti razmatranog podgrafa su putevi ili ciklusi. Budući da svaki ciklus sadrži jednak broj grana iz sparivanja $M$ i $M'$ (ciklus mora biti parne dužine i njegove grane naizmenično pripadaju jednom pa drugom sparivanju) i da sparivanje $M'$ sadrži više grana, zaključujemo da mora postojati komponenta koja je put koji sadrži jednu granu više iz $M'$ nego iz $M$. U tom slučaju, grane takvog puta naizmenično pripadaju sparivanjima $M$ i $M'$, s tim da prva i poslednja grana puta pripadaju sparivanju $M'$. Taj put je, dakle, traženi $M$-uvećani put.
\end{proof}

\subsection*{Problem rasporeda poslova (Asignacija)}
Problem se modeluje pomoću bipartitnog grafa gde jedna particija čvorova predstavlja radnike, a druga poslove.

\begin{itemize}
    \item \textbf{Jednostavan slučaj:} Ako su svi radnici jednako efikasni, problem se svodi na pronalaženje \textbf{najvećeg sparivanja} u bipartitnom grafu. Za to se koristi \textbf{Mađarski algoritam}.
    \item \textbf{Složeniji slučaj:} Ako radnici imaju različitu efikasnost, koriste se težinski bipartitni grafovi gde težina grane predstavlja cenu ili vreme obavljanja posla. Ovde se primenjuje \textbf{Kun-Mankresov algoritam}.
\end{itemize}

\section{Bojenje grafova. Problem 4 boje.}
Bojenje grafova podrazumeva dodeljivanje boja elementima grafa (čvorovima ili granama) uz određene uslove.

\begin{itemize}
    \item \textbf{Pravilno bojenje čvorova} - funkcija kojom se susednim čvorovima dodeljuju različite boje.
    \item \textbf{Hromatski broj $\chi(G)$} - minimalan broj boja potrebnih za pravilno bojenje čvorova. To je invarijanta grafa.
    \item \textbf{Linijski graf $L(G)$} - graf čiji čvorovi odgovaraju granama grafa $G$, a susedni su ako su odgovarajuće grane u $G$ susedne. Bojenje grana grafa $G$ ekvivalentno je bojenju čvorova grafa $L(G)$.
\end{itemize}

\begin{itemize}
    \item \textbf{Kompletni grafovi:} $\chi(K_n) = n$.
    \item \textbf{Bipartitni grafovi:} Za graf sa bar jednom granom, $\chi(G) = 2$ (bihromatski grafovi).
    \item \textbf{Ciklus $C_n$:} $\chi(C_n) = 2$ za $n=2k$, odnosno $\chi(C_n) = 3$ za $n=2k-1$.
    \item \textbf{Gornja granica:} $\chi(G) \leq \Delta + 1$.
    \item \textbf{Bruks teorema:} Jednakost $\chi(G) = \Delta + 1$ važi akko je graf kompletan ili ciklus neparne dužine.
\end{itemize}

\subsection{Bojenje mapa}
Problem bojenja mapa datira iz XIX veka i odnosi se na bojenje teritorija tako da susedne teritorije (one koje dele granicu dužine veće od nule) imaju različite boje.

\begin{itemize}
    \item \textbf{Modelovanje:} Svaka mapa se može predstaviti kao planaran graf gde čvorovi odgovaraju teritorijama, a grane granicama. Problem se svodi na određivanje hromatskog broja tog planarnog grafa.
    \item \textbf{Teorema o četiri boje:} Svaki planaran graf je 4-obojiv ($\chi(G) \leq 4$).
    \item \textbf{Istorijat:} Hipotezu je 1890. godine podržao Hejvud (dokazao 5-obojivost), dok su Epl i Haken tek 1976. godine dokazali teoremu o 4 boje uz pomoć računara.
\end{itemize}


\subsection*{Primene i algoritmi}
\begin{itemize}
    \item \textbf{Pohlepna heuristika:} Čvorovi se boje redom ($v_1, \dots, v_n$), a svakom se dodeljuje prva boja koja nije iskorišćena za njegove susede. Iako efikasan, ovaj algoritam ne garantuje minimalan broj boja ($\chi$).
    \item \textbf{Raspored ispita:} Ispiti su čvorovi, a grane povezuju one koje student polaže istovremeno. Potreban broj termina = $\chi(G)$.
    \item \textbf{Skladištenje materijala:} Materije su čvorovi, a grane povezuju nekompatibilne supstance. Minimalan broj skladišnih prostorija = $\chi(G)$.
\end{itemize}
